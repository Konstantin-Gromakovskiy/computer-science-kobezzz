## Производительность операций с массивами в JS

Необходимо исследовать разницу в скорости между `push`/`pop` и `unshift`/`shift`, а также влияние «дырок» на производительность.

Сравните скорость операций для массивов разного размера (10, 100, 1000 и так далее).

**Что измерить:**

- `push` — добавление в конец;
- `pop` — удаление из конца;
- `unshift` — добавление в начало;
- `shift` — удаление из начала.

Проделайте тесты для массивов с дырками и без. Не забудьте сделать «прогрев JIT» перед замерами.

Проанализируйте результат и сделайте выводы.

const create = {
  packed(size) {
    return Array.from({ length: size }, (_, i) => i);
  },

  holey(size) {
    const arr = new Array(size);

    for (let i = 0; i < size; i++) {
      arr[i] = i;
    }

    return arr;
  },

  sparse(size, fillRatio = 0.3) {
    const arr = new Array(size);

    arr[0] = 0;

    if (size > 1) {
      arr[size - 1] = size - 1;
    }

    for (let i = 1; i < size - 1; i++) {
      if (Math.random() < fillRatio) {
        arr[i] = i;
      }
    }

    return arr;
  },
};

function getPercentiles(times) {
  const sorted = [...times].sort((a, b) => a - b);
  const p50 = sorted[Math.floor(sorted.length * 0.5)];
  const p95 = sorted[Math.floor(sorted.length * 0.95)];
  const p99 = sorted[Math.floor(sorted.length * 0.99)];
  const mean = times.reduce((a, b) => a + b, 0) / times.length;
  return { mean, p50, p95, p99 };
}

function measureWithStats(fn, iterations = 5000) {
  // Прогрев JIT
  for (let i = 0; i < 1000; i++) {
    fn();
  }

  globalThis.gc?.();

  const times = [];

  for (let i = 0; i < iterations; i++) {
    const start = performance.now();
    fn();
    const end = performance.now();
    times.push((end - start) * 1000); // В микросекундах
  }

  return getPercentiles(times);
}

function getMark(ratio) {

}
